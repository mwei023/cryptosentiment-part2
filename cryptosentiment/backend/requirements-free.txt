# Lightweight dependency set for Render's FREE web service tier (512 MB RAM).
#
# This is requirements.txt minus the heavy ML stack:
#   - torch / transformers  (~1.5+ GB installed, model loads ~1-2 GB RAM)
#   - prophet               (brings cmdstanpy + compiled stan models)
#
# The API starts with DISABLE_HEAVY_MODELS=1 (set in render.yaml) so main.py
# returns 503 on sentiment endpoints and prediction confidence degrades to
# price-only; price/paper-trade/research endpoints are unaffected.
#
# For a paid instance with FinBERT enabled (DISABLE_HEAVY_MODELS unset),
# deploy with the full requirements.txt instead.

# --- API layer ---
fastapi
uvicorn
gunicorn

# --- Database ---
sqlalchemy
psycopg2-binary

# --- Data ---
pandas
numpy

# --- HTTP clients & config ---
requests
httpx
python-dotenv
